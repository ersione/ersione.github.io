CI 워크플로우 — PR 단계

# .github/workflows/ci.yml
name: CI
 
on:
  pull_request:
    branches: [main, develop]
 
jobs:
  test:
    runs-on: ubuntu-latest
    services:
      mysql:
        image: mysql:8.0
        env:
          MYSQL_ROOT_PASSWORD: test
          MYSQL_DATABASE: testdb
        options: --health-cmd="mysqladmin ping" --health-interval=10s
 
    steps:
      - uses: actions/checkout@v4
 
      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          java-version: '21'
          distribution: 'temurin'
          cache: gradle
 
      - name: Run Tests
        run: ./gradlew test --parallel
        env:
          SPRING_DATASOURCE_URL: jdbc:mysql://localhost:3306/testdb
 
      - name: Code Coverage Report
        run: ./gradlew jacocoTestReport
 
      - name: Upload Coverage to Codecov
        uses: codecov/codecov-action@v3
 
      - name: SonarQube Analysis
        uses: SonarSource/sonarcloud-github-action@master
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}

CD 워크플로우 — 프로덕션 배포

# .github/workflows/cd-prod.yml
name: CD - Production
 
on:
  push:
    branches: [main]
 
jobs:
  deploy:
    runs-on: ubuntu-latest
    environment: production  # GitHub Environment 보호 규칙 적용
 
    steps:
      - uses: actions/checkout@v4
 
      - name: Configure AWS Credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ap-northeast-2
 
      - name: Build & Push Docker Image to ECR
        run: |
          IMAGE_TAG=${{ github.sha }}
          
          # ECR 로그인
          aws ecr get-login-password | docker login --username AWS \
            --password-stdin $ECR_REGISTRY
          
          # 멀티스테이지 빌드 (캐시 활용)
          docker buildx build \
            --cache-from $ECR_REGISTRY/order-service:cache \
            --cache-to $ECR_REGISTRY/order-service:cache \
            --tag $ECR_REGISTRY/order-service:$IMAGE_TAG \
            --push .
 
      - name: Trivy Security Scan
        uses: aquasecurity/trivy-action@master
        with:
          image-ref: "${{ env.ECR_REGISTRY }}/order-service:${{ github.sha }}"
          severity: 'CRITICAL,HIGH'
          exit-code: '1'  # CRITICAL 취약점 시 배포 중단
 
      - name: Deploy to Kubernetes (Canary)
        run: |
          # Argo Rollouts로 Canary 배포 시작
          kubectl argo rollouts set image order-rollout \
            order-service=$ECR_REGISTRY/order-service:${{ github.sha }}
          
          # 배포 모니터링 (5분 대기)
          kubectl argo rollouts status order-rollout --timeout=5m

Dockerfile 최적화

# 멀티스테이지 빌드로 이미지 크기 최소화
FROM eclipse-temurin:21-jdk-alpine AS builder
WORKDIR /app
COPY gradlew .
COPY gradle gradle
COPY build.gradle.kts settings.gradle.kts ./
# 의존성 레이어 캐시 (코드 변경 시 재사용)
RUN ./gradlew dependencies --no-daemon
 
COPY src src
RUN ./gradlew bootJar --no-daemon
 
# 최종 이미지 (JRE만 포함)
FROM eclipse-temurin:21-jre-alpine
WORKDIR /app
 
# 보안: non-root 사용자
RUN addgroup -S spring && adduser -S spring -G spring
USER spring:spring
 
# Spring Boot Layered JAR 활용
COPY --from=builder /app/build/libs/*.jar app.jar
 
EXPOSE 8080
ENTRYPOINT ["java", \
  "-XX:+UseContainerSupport", \
  "-XX:MaxRAMPercentage=75.0", \
  "-jar", "app.jar"]

멀티스테이지 빌드 효과: 이미지 크기 1.2GB → 180MB (85% 감소). ECR 푸시/풀 시간이 크게 줄어 전체 파이프라인이 빨라집니다.
